#wasserstein distance in 2D
import numpy as np
import ot  # pip install pot

def wasserstein_dist_cnn(x, y):
    """
    Calcola la distanza di Wasserstein tra due feature maps da CNN.
    
    Parametri:
        x, y : array numpy con shape (C, H, W)
    
    Ritorna:
        distanza (float)
    """
    # Reshape: ogni posizione spaziale diventa un punto in R^C
    C, H, W = x.shape
    X = x.reshape(C, -1).T   # shape (H*W, C)
    Y = y.reshape(C, -1).T   # shape (H*W, C)

    # Distribuzioni di probabilità uniformi (o pesate dai valori)
    a = np.ones(X.shape[0]) / X.shape[0]
    b = np.ones(Y.shape[0]) / Y.shape[0]

    # Matrice dei costi (distanze euclidee tra vettori in R^C)
    M = ot.dist(X, Y, metric="euclidean")

    # Calcolo della distanza di Wasserstein (Earth Mover's Distance)
    dist = ot.emd2(a, b, M)
    return dist

# 🔹 Esempio di utilizzo
fmap1 = np.random.rand(16, 8, 8)   # 16 canali, 8x8 spatial
fmap2 = np.random.rand(16, 8, 8)

d = wasserstein_dist_cnn(fmap1, fmap2)
print("Distanza di Wasserstein CNN:", d)
